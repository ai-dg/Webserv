server {
    listen 8090;
    host 127.0.0.2;

    server_name projects.42.fr;

    error_page_403 /default/error_pages/403.html;
    error_page_404 /default/error_pages/404.html;
    error_page_500 /default/error_pages/500.html;

#   set (K)ilo-bytes (M)ega-bytes
    client_max_body_size 1000M;

    keepalive_timeout 65;
    client_body_timeout 60;
    client_header_timeout 10;

    location / {
        root /www/html/projects42;
        index index.html;
        methods GET POST;
    }

    location /images/ {
        root /www/html/projects42/images;
        extension .png .jpg .gif .jpeg;
        methods GET;
    }

    location /upload/ {
        root /www/uploads;
        methods GET POST DELETE;
        autoindex on;
    }

    location /cgi-bin/ {
        root /cgi-bin;
        cgi on;
        cgi_bin /cgi-bin/;
        methods GET POST;
        extension .py .pl .sh .php;
    }

    location /test/ {
        return 301 https://www.google.fr;
    }

    location /test2/ {
        return 301 /;
    }

        return 301 http://127.0.0.5:9000;
